  const { applicationNumber, newTeacherId, shareDetails, StudentshareDetails } =
    req.body;
  if (
    !applicationNumber ||
    !newTeacherId ||
    !shareDetails ||
    !StudentshareDetails
  ) {
    return res.status(400).json({
      message:
        "All fields (applicationNumber, newTeacherId, shareDetails, StudentshareDetails) are required.",
    });
  }

  try {
    const interview = await InterviewRequest.findOne({ applicationNumber });

    if (!interview) {
      return res.status(404).json({ message: "Application not found." });
    }

    // Add the new teacher to the interview request
    interview.teacher.push({
      teacherId: newTeacherId,
      shareDetails,
      StudentshareDetails,
      status: "Pending",
    });

    // Save the updated interview request
    await interview.save();

    // Notify the new teacher (you can implement a notification logic here)
    const teacher = await TeacherProfile.findById(newTeacherId);
    if (teacher) {
      teacher.notifications.push({
        type: "Shared Interview Request",
        applicationNumber,
        details: { shareDetails, StudentshareDetails },
      });

      await teacher.save();
    }

    res.status(200).json({
      message: "Application shared successfully.",
      success: true,
    });
  } catch (error) {
    console.error("Error sharing interview request:", error.message);
    res.status(500).json({ message: "Server error: " + error.message });
  }

const shareInterviewRequest = async (req, res) => {
  const { applicationNumber, newTeacherId, shareDetails, StudentshareDetails } =
    req.body;

  console.log("Received request body:", req.body);

  if (
    !applicationNumber ||
    !newTeacherId ||
    !shareDetails ||
    !StudentshareDetails
  ) {
    console.log("Missing required fields:", {
      applicationNumber,
      newTeacherId,
      shareDetails,
      StudentshareDetails,
    });
    return res.status(400).json({
      message:
        "Missing required fields: applicationNumber, newTeacherId, shareDetails, StudentshareDetails.",
    });
  }

  try {
    const interview = await InterviewRequest.findOne({ applicationNumber });
    console.log("Fetched interview:", interview);

    if (!interview) {
      console.log(
        "Application not found for applicationNumber:",
        applicationNumber
      );
      return res.status(404).json({ message: "Application not found." });
    }

    if (interview.status === "Shared") {
      console.log("Application already shared:", applicationNumber);
      return res
        .status(400)
        .json({ message: "This application has already been shared." });
    }

    const existingTeacher = interview.teacher.find(
      (teacher) => teacher.teacherId.toString() === newTeacherId
    );

    if (existingTeacher) {
      console.log("Teacher already associated with this application:", {
        newTeacherId,
        applicationNumber,
      });
      return res.status(400).json({
        message:
          "This application is already shared with the selected teacher.",
      });
    }

    interview.teacher.push({
      teacherId: newTeacherId,
      shareDetails,
      StudentshareDetails,
      status: "Shared",
    });

    interview.status = "Shared"; // Update global status
    await interview.save();

    // Log the updated interview object to ensure the changes are saved
    const updatedInterview = await InterviewRequest.findOne({
      applicationNumber,
    });
    console.log("Updated interview status:", updatedInterview.status); // Log updated status

    const teacher = await TeacherProfile.findById(newTeacherId);
    if (teacher) {
      teacher.notifications.push({
        type: "Shared Interview Request",
        applicationNumber,
        details: { shareDetails, StudentshareDetails },
      });
      await teacher.save();
      console.log("Teacher notified:", teacher); // Log teacher notification
    }

    res.status(200).json({
      message: "Application shared successfully.",
      success: true,
    });
  } catch (error) {
    console.error("Error in shareInterviewRequest:", error.message);
    res.status(500).json({ message: "Server error: " + error.message });
  }
};
